\begin{tcolorbox}[colback=white,colframe=tealUteq,boxrule=0.8pt,arc=2pt,left=6pt,right=6pt,top=5pt,bottom=5pt]
\footnotesize
\[ \text{Puntos}_{i} = \text{peso}_{i}\times \frac{\text{nivel}_{i}}{4} \]
\[ \text{Nota}_{/10} = \frac{1}{10}\sum_{i} \text{peso}_{i}\times \frac{\text{nivel}_{i}}{4} \]
con $\text{peso}_i$ en \% ($\sum \text{peso}_i = 100$) y $\text{nivel}_i \in \{0,1,2,3,4\}$. El total sobre 100 equivale a la nota sobre 10.
\end{tcolorbox}

% ================= CONDICIONES ELIMINATORIAS =================
\begin{tcolorbox}[colback=rojoElim!6,colframe=rojoElim,boxrule=1pt,arc=2pt,left=7pt,right=7pt,top=4pt,bottom=4pt]
\footnotesize
\textbf{\textcolor{rojoElim}{CONDICIONES ELIMINATORIAS -- la nota global de la practica es 0 si se cumple cualquiera de estas:}}
\begin{enumerate}[leftmargin=1.4em,itemsep=1pt,topsep=2pt]
\item El documento \textbf{no contiene la URL del repositorio GitHub}.
\item El \textbf{ultimo commit no corresponde a la fecha y hora de entrega} de la tarea, o el repositorio es inexistente/privado/inaccesible.
\item Existe \textbf{una o mas referencias inventadas, fabricadas o no verificables} (DOI/ISBN inexistente o que no corresponde al titulo): se interpreta como uso de IA no declarado y se penaliza drasticamente.
\item El sistema \textbf{no es demostrable en vivo}: \texttt{docker-compose up -{}-build} falla o los endpoints no responden.
\item \textbf{Ausencia total de un componente obligatorio} del sistema: los tres microservicios, el API Gateway (Nginx), el archivo \texttt{docker-compose.yml} o el documento LaTeX.
\end{enumerate}
\end{tcolorbox}
